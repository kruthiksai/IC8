pipeline {
    agent { label 'DevServer' }

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main', url: 'https://github.com/kruthiksai/IC8'
            }
        }

        stage('Install Dependencies') {
            steps {
                sh '''
                    sudo apt update
                    sudo apt install -y python3-flask python3-flask-cors
                '''
            }
        }

        stage('Run Flask App') {
            steps {
                sh '''
                    cd $WORKSPACE
                    pkill -f app.py || echo "No running Flask app"

                    nohup python3 app.py > flask.log 2>&1 &
                '''
            }
        }
    }
}
